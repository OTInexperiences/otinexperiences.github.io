Not sure how many switches I need so started out with an array of 5 by 4.  The switches are connected in a 5 x 4 grid so that all switches on  each row have a common input connection (all the black wires) and all switches on each column have a common output connection (all the coloured wires). So  for example the switch under my finger will only be powered when a voltage is placed on the bottom black wire. When that occurs pressing the switch will cause the voltage to be transferred only to the corresponding column wire  which is yellow in this example.
<p style="text-align: left;">You need to scan the key array in order to establish which key if any has been pressed. This is achieved by sequentially setting each rows voltage high and then checking for a voltage on each column.</p>
<p style="text-align: left;">The code I have written so far places the keyboard object in its own thread so that scanning the keyboard is independant of the rest of the main program. When a key is pressed an event is generated. Done this way it removes the necessity of having the main program continually polling the keyboard.</p>
<p style="text-align: left;">It is connected to my FEZ Panda2 processor board, the code for the class is shown below</p>
<pre>using System;
using System.Threading;
using Microsoft.SPOT;
using Microsoft.SPOT.Hardware;
using GHIElectronics.NETMF.FEZ;

namespace FEZ_Panda_II_Application1
{
    public class KeyPressEventArgs : EventArgs
    {
        public char Key { get; set; }
        public KeyPressEventArgs(char c)
        {
            Key = c;
        }
    }

    class keyboard
    {
        OutputPort[] rows = new OutputPort[5]
        {
            new OutputPort((Cpu.Pin)FEZ_Pin.Digital.Di8,false),
            new OutputPort((Cpu.Pin)FEZ_Pin.Digital.Di9,false),
            new OutputPort((Cpu.Pin)FEZ_Pin.Digital.Di10,false),
            new OutputPort((Cpu.Pin)FEZ_Pin.Digital.Di11,false),
            new OutputPort((Cpu.Pin)FEZ_Pin.Digital.Di13,false)
        };
        InputPort[] cols = new InputPort[4]
        {
            new InputPort((Cpu.Pin)FEZ_Pin.Digital.Di4, false, Port.ResistorMode.PullDown),
            new InputPort((Cpu.Pin)FEZ_Pin.Digital.Di5, false, Port.ResistorMode.PullDown),
            new InputPort((Cpu.Pin)FEZ_Pin.Digital.Di6, false, Port.ResistorMode.PullDown),
            new InputPort((Cpu.Pin)FEZ_Pin.Digital.Di7, false, Port.ResistorMode.PullDown),
        };
        char[][] keys =
        {
            new char[] {'1','2','3','4'},
            new char[] {'5','6','7','8'},
            new char[] {'9','0','A','B'},
            new char[] {'C','D','E','F'},
            new char[] {'G','H','I','J'}
        };

        #region Public Event
        public delegate void KeyPressEventDelegate(object sender, KeyPressEventArgs e);
        public event KeyPressEventDelegate KeypressEvent;

        private void OnKeyPressEvent(KeyPressEventArgs e)
        {
            if (KeypressEvent != null)
                KeypressEvent(this, e);
        }
        #endregion

        public char Scan()
        {
            char ch = 'X';
            for (int row = 0; row &lt; 5; row++)
            {
                rows[row].Write(true);
                Thread.Sleep(50);
                for (int col = 0; col &lt; 4; col++)
                {
                    if (cols[col].Read())
                        ch = keys[row][col];
                }
                rows[row].Write(false);
            }
            return ch;
        }

        public keyboard()
        {
            ThreadStart start = delegate()
            {
                while (true)
                {
                    char c = Scan();
                    if (c != 'X')
                        OnKeyPressEvent(new KeyPressEventArgs(c));
                    Thread.Sleep(100);
                }
            };
            new Thread(start).Start();
        }
    }
}</pre>
The test program code is
<pre>using System;
using System.Threading;
using Microsoft.SPOT;
using Microsoft.SPOT.Hardware;
using GHIElectronics.NETMF.FEZ;

namespace FEZ_Panda_II_Application1
{
    public class Program
    {
        static ScreenManager screen_manager = new ScreenManager();
        static keyboard kbd = new keyboard();

        public static void Main()
        {
            // Blink board LED
            bool ledState = false;
            string currkey = string.Empty;
            OutputPort led = new OutputPort((Cpu.Pin)FEZ_Pin.Digital.LED, ledState);
            kbd.KeypressEvent +=new keyboard.KeyPressEventDelegate(kbd_KeypressEvent);

            int n = 0;
            while (true)
            {
                screen_manager.Clear();
                screen_manager.Write(n.ToString());
                n++;
                Thread.Sleep(1000);
                // toggle LED state
                ledState = !ledState;
                led.Write(ledState);
            }
        }

        public static void kbd_KeypressEvent(object sender, KeyPressEventArgs e)
        {
            screen_manager.Write(" Key = " + e.Key.ToString());
        }
    }
}</pre>
